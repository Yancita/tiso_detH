Dz=1
E=0
Di(B,Dz,E)=(3*B*B+Dz*Dz+3.*E*E)**3 - (Dz**3-9*Dz*E*E)**2
q(Dz,E) = Dz**3-9*Dz*E*E
I={0.,1.}
w=exp(2*pi*I/3.)
u2(B,Dz,E) = ((-q(Dz,E)+I*sqrt(Di(B,Dz,E)))**(1./3.)+(-q(Dz,E)-I*sqrt(Di(B,Dz,E)))**(1./3.))/3.
u1(B,Dz,E) = (w**2*(-q(Dz,E)+I*sqrt(Di(B,Dz,E)))**(1./3.)+w*(-q(Dz,E)-I*sqrt(Di(B,Dz,E)))**(1./3.))/3.
u0(B,Dz,E) = (w*(-q(Dz,E)+I*sqrt(Di(B,Dz,E)))**(1./3.)+w**2*(-q(Dz,E)-I*sqrt(Di(B,Dz,E)))**(1./3.))/3.


l2(B,Dz,E) = u2(B,Dz,E) + 4.*B**4*(Dz*Dz+3.*E*E)*u2(B,Dz,E)/(15*(u1(B,Dz,E)-u2(B,Dz,E))**3*((u0(B,Dz,E)-u2(B,Dz,E)))**3)
l1(B,Dz,E) = u1(B,Dz,E) + 4.*B**4*(Dz*Dz+3.*E*E)*u1(B,Dz,E)/(15*(u2(B,Dz,E)-u1(B,Dz,E))**3*((u0(B,Dz,E)-u1(B,Dz,E)))**3)
l0(B,Dz,E) = u0(B,Dz,E) + 4.*B**4*(Dz*Dz+3.*E*E)*u0(B,Dz,E)/(15*(u1(B,Dz,E)-u0(B,Dz,E))**3*((u2(B,Dz,E)-u0(B,Dz,E)))**3)


PsiZ0(x)= 1./(1.+(1./(Dz*Dz)+3.*E*E/(Dz*Dz*Dz*Dz))*x*x/(1.+(0.24/(Dz*Dz)+1.6*E*E/(Dz*Dz*Dz*Dz))*x*x/(1.+(0.6/(Dz)+2.2*E*E/(Dz*Dz*Dz))*abs(x))))
PsiZ2(x)=1./(1.+(1./(Dz+E)**2)*x*x/(1.+(0.8/(Dz+E)**2)*x*x/(1.+(1/Dz)*abs(x))))

Sigma0Z(x) = (1 - PsiZ0(x))*(2./3.)
Sigma2Z(x) = (2 + PsiZ2(x))*(1./3.)
Sigma1Z(x) = (2 - Sigma0Z(x) - Sigma2Z(x))


p0(B, T) = exp(-l0(B, Dz, E)/T) / (exp(-l0(B, Dz, E)/T)+exp(-l1(B, Dz, E)/T)+exp(-l2(B, Dz, E)/T))
p1(B, T) = exp(-l1(B, Dz, E)/T) / (exp(-l0(B, Dz, E)/T)+exp(-l1(B, Dz, E)/T)+exp(-l2(B, Dz, E)/T))
p2(B, T) = exp(-l2(B, Dz, E)/T) / (exp(-l0(B, Dz, E)/T)+exp(-l1(B, Dz, E)/T)+exp(-l2(B, Dz, E)/T))
PLtemp(B, T) = p0(B,T) * Sigma0Z(B) + p1(B,T) * Sigma1Z(B) + p2(B,T) * Sigma2Z(B)
